// ENTWURF (nicht aktiv). Wird tests/workflow-invariants.test.ts. Braucht devDependency "yaml".
// Sperrt die Eigenschaften, die M0-04 sichern soll. Hinweis: Ein Pull Request kann diesen Test selbst ändern,
// deshalb ist er nur ein Frühwarnsystem. Die eigentliche Sperre sind Umgebungsregel + Regelsatz auf "deploy" (EINRICHTUNG.md, Probe).
import { readFileSync, readdirSync } from "node:fs";
import { parse } from "yaml";
import { describe, expect, it } from "vitest";

const dir = ".github/workflows";
const files = readdirSync(dir).filter((f) => /\.ya?ml$/.test(f));
const load = (f: string) => parse(readFileSync(`${dir}/${f}`, "utf8"));
const jobsOf = (wf: any) => Object.entries<any>(wf.jobs ?? {});
const text = (f: string) => readFileSync(`${dir}/${f}`, "utf8");

describe("Workflows: kein Schreibrecht für Pull-Request-Läufe", () => {
  it("es gibt mindestens einen Workflow", () => expect(files.length).toBeGreaterThan(0));

  for (const f of files) {
    const wf = load(f);
    it(`${f}: Workflow-Rechte sind read-only und ohne write`, () => {
      const p = wf.permissions;
      expect(p, "permissions muss oben gesetzt sein").toBeDefined();
      expect(p).not.toBe("write-all");
      for (const v of Object.values<string>(typeof p === "object" ? p : {})) expect(v).not.toBe("write");
    });
    it(`${f}: kein Job hat Schreibrechte`, () => {
      for (const [name, job] of jobsOf(wf))
        for (const v of Object.values<string>(job.permissions ?? {})) expect(v, `${f}:${name}`).not.toBe("write");
    });
    it(`${f}: kein Schreibzugriff über GITHUB_TOKEN-Push`, () => {
      expect(text(f)).not.toMatch(/x-access-token/);
      expect(text(f)).not.toMatch(/secrets\.GITHUB_TOKEN[\s\S]{0,200}git push|git push[\s\S]{0,200}secrets\.GITHUB_TOKEN/);
    });
    it(`${f}: Aktionen sind auf Commit-SHA festgelegt`, () => {
      for (const m of text(f).matchAll(/uses:\s*([^\s#]+)/g)) if (!m[1].startsWith("./")) expect(m[1]).toMatch(/@[0-9a-f]{40}$/);
    });
    it(`${f}: Checkout speichert keine Zugangsdaten`, () => {
      for (const [name, job] of jobsOf(wf))
        for (const s of job.steps ?? [])
          if (String(s.uses ?? "").startsWith("actions/checkout")) expect(s.with?.["persist-credentials"], `${f}:${name}`).toBe(false);
    });
  }
});

describe("deploy.yml: nur der Auslieferungs-Job schreibt, nur von main aus", () => {
  const wf = load("deploy.yml");
  const jobs = Object.fromEntries(jobsOf(wf));
  it("Job 'ausliefern' nutzt die Umgebung 'auslieferung' und läuft nur bei Push auf main", () => {
    expect(jobs.ausliefern.environment).toBe("auslieferung");
    expect(jobs.ausliefern.if).toContain("github.event_name == 'push'");
    expect(jobs.ausliefern.if).toContain("refs/heads/main");
    expect(jobs.ausliefern.needs).toBe("build");
  });
  it("das Deploy-Geheimnis kommt nur in 'ausliefern' vor", () => {
    for (const [name, job] of Object.entries<any>(jobs))
      if (name !== "ausliefern") expect(JSON.stringify(job)).not.toContain("DEPLOY_KEY");
  });
  it("Push auf 'deploy' steht nur in 'ausliefern'", () => {
    for (const [name, job] of Object.entries<any>(jobs))
      if (name !== "ausliefern") expect(JSON.stringify(job)).not.toMatch(/git push/);
  });
  it("Pull-Request-Läufe erreichen 'ausliefern' nie", () => {
    expect(wf.on.pull_request).not.toBeUndefined();
    expect(jobs.ausliefern.if).not.toMatch(/pull_request/);
  });
});
